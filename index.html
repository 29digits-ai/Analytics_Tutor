    4. Spaced Repetition Engine (SM-2)

    Implements the SuperMemo-2 algorithm — the same engine behind Anki. Each topic becomes an SRS card graded on the 0–5 scale, automatically scheduling reviews at 1, 3, 7, 14, 30 day intervals (and beyond) with a dynamically adjusted ease factor.

    lib/srs.ts
    // SM-2 algorithm (SuperMemo 2) implementation
export interface SRSCard {
  topicId: string;
  trackId: string;
  moduleId: string;
  easeFactor: number;      // starting at 2.5
  interval: number;        // days
  repetitions: number;
  dueDate: number;         // epoch ms
  lastReview: number;
}

export type Quality = 0 | 1 | 2 | 3 | 4 | 5; // 0=blackout, 5=perfect

export function grade(card: SRSCard, q: Quality): SRSCard {
  const next: SRSCard = { ...card, lastReview: Date.now() };

  if (q < 3) {
    next.repetitions = 0;
    next.interval = 1;
  } else {
    if (next.repetitions === 0) next.interval = 1;
    else if (next.repetitions === 1) next.interval = 3;
    else if (next.repetitions === 2) next.interval = 7;
    else if (next.repetitions === 3) next.interval = 14;
    else if (next.repetitions === 4) next.interval = 30;
    else next.interval = Math.round(next.interval * next.easeFactor);
    next.repetitions += 1;
  }

  // Update ease factor
  next.easeFactor = Math.max(
    1.3,
    next.easeFactor + (0.1 - (5 - q) * (0.08 + (5 - q) * 0.02))
  );

  const dayMs = 24 * 60 * 60 * 1000;
  next.dueDate = Date.now() + next.interval * dayMs;
  return next;
}

export function getDueCards(cards: SRSCard[]): SRSCard[] {
  const now = Date.now();
  return cards.filter(c => c.dueDate <= now).sort((a, b) => a.dueDate - b.dueDate);
}

    5. Storage Layer (IndexedDB)

    A thin wrapper around the idb library. All user state — notes, SRS cards, progress, LLM keys — lives locally with automatic JSON export/import backup support.

    lib/storage.ts
    import { openDB, DBSchema, IDBPDatabase } from 'idb';
import type { SRSCard } from './srs';

interface AppDB extends DBSchema {
  notes: { key: string; value: { topicId: string; text: string; updatedAt: number } };
  srs:   { key: string; value: SRSCard };
  progress: { key: string; value: { topicId: string; completed: boolean; score?: number; updatedAt: number } };
  settings: { key: string; value: { apiKey?: string; endpoint?: string } };
}

let dbPromise: Promise<IDBPDatabase<AppDB>>;
function getDB() {
  if (!dbPromise) {
    dbPromise = openDB<AppDB>('ai-tutor', 1, {
      upgrade(db) {
        db.createObjectStore('notes',     { keyPath: 'topicId' });
        db.createObjectStore('srs',       { keyPath: 'topicId' });
        db.createObjectStore('progress',  { keyPath: 'topicId' });
        db.createObjectStore('settings',  { keyPath: 'id' } as any);
      }
    });
  }
  return dbPromise;
}

export const storage = {
  saveNote:  async (topicId: string, text: string) => (await getDB()).put('notes', { topicId, text, updatedAt: Date.now() }),
  loadNote:  async (topicId: string) => (await getDB()).get('notes', topicId),
  saveSRS:   async (card: SRSCard) => (await getDB()).put('srs', card),
  allSRS:    async () => (await getDB()).getAll('srs'),
  markDone:  async (topicId: string, score?: number) => (await getDB()).put('progress', { topicId, completed: true, score, updatedAt: Date.now() }),
  getProgress: async () => (await getDB()).getAll('progress'),
  saveKey:   async (apiKey: string, endpoint?: string) => (await getDB()).put('settings', { id: 'llm', apiKey, endpoint } as any),
  loadKey:   async () => (await getDB()).get('settings', 'llm' as any),
};

    6. Dashboard

    The home route shows overall mastery, the daily study streak, the SRS review queue, and cards for all 5 learning tracks.

    app/page.tsx
    'use client';
import { useEffect, useState } from 'react';
import Link from 'next/link';
import { storage } from '@/lib/storage';
import { getDueCards, SRSCard } from '@/lib/srs';
import { tracks } from '@/lib/curriculum';
import { Flame, BookOpen, Calendar, ChevronRight, RotateCcw } from 'lucide-react';

export default function Dashboard() {
  const [due, setDue] = useState<SRSCard[]>([]);
  const [progress, setProgress] = useState<any[]>([]);
  const [streak, setStreak] = useState(0);

  useEffect(() => {
    (async () => {
      const cards = await storage.allSRS();
      setDue(getDueCards(cards));
      setProgress(await storage.getProgress());
      const last = localStorage.getItem('lastStudy');
      const today = new Date().toDateString();
      if (last === today) setStreak(parseInt(localStorage.getItem('streak') || '1'));
      else { localStorage.setItem('lastStudy', today); setStreak(1); localStorage.setItem('streak', '1'); }
    })();
  }, []);

  const totalTopics = 51 * 6; // approximate
  const pct = Math.round((progress.length / totalTopics) * 100);

  return (
    <div className="max-w-7xl mx-auto p-8">
      <h1 className="text-3xl font-bold mb-2">Welcome back</h1>
      <p className="text-zinc-400 mb-8">Continue your 6-step active learning journey.</p>

      {/* Hero metrics */}
      <div className="grid grid-cols-1 md:grid-cols-4 gap-4 mb-10">
        <MetricCard icon={<Flame className="text-orange-400" />} label="Study Streak" value={`${streak} days`} />
        <MetricCard icon={<BookOpen className="text-blue-400" />} label="Topics Mastered" value={progress.length} />
        <MetricCard icon={<Calendar className="text-emerald-400" />} label="Due Today" value={due.length} tone={due.length > 0 ? 'warn' : 'ok'} />
        <MetricCard icon={<RotateCcw className="text-purple-400" />} label="Overall Mastery" value={`${pct}%`} />
      </div>

      {/* Due reviews */}
      {due.length > 0 && (
        <section className="mb-10">
          <h2 className="text-xl font-semibold mb-4 flex items-center gap-2">
            <Calendar className="w-5 h-5" /> Due for Review
          </h2>
          <div className="grid grid-cols-1 md:grid-cols-3 gap-3">
            {due.slice(0, 6).map(card => (
              <Link key={card.topicId}
                href={`/track/${card.trackId}/module/${card.moduleId}/topic/${card.topicId}`}
                className="p-4 rounded-lg bg-amber-500/10 border border-amber-500/30 hover:bg-amber-500/20 transition">
                <div className="text-sm text-amber-300 font-medium">{card.moduleId}</div>
                <div className="font-semibold">{card.topicId}</div>
                <div className="text-xs text-zinc-400 mt-1">Interval: {card.interval}d · EF: {card.easeFactor.toFixed(2)}</div>
              </Link>
            ))}
          </div>
        </section>
      )}

      {/* Tracks */}
      <section>
        <h2 className="text-xl font-semibold mb-4">Your Learning Tracks</h2>
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
          {tracks.map(t => (
            <Link key={t.id} href={`/track/${t.id}`}
              className="group p-6 rounded-xl bg-zinc-900 border border-zinc-800 hover:border-zinc-700 hover:shadow-xl hover:shadow-blue-500/10 transition">

    9. Monaco + Pyodide Sandbox

    In-browser coding for statistics, data analysis, and any technical module. Monaco provides the editor; Pyodide runs real Python (with numpy, pandas, scipy) in WebAssembly — zero server cost.

    components/editor/PyodideRunner.ts
    export class PyodideRunner {
  private ready: Promise<any> | null = null;
  async init() {
    if (this.ready) return this.ready;
    this.ready = (async () => {
      // @ts-ignore
      const pyodide = await import('pyodide').then(m => m.loadPyodide({ indexURL: 'https://cdn.jsdelivr.net/pyodide/v0.26.2/full/' }));
      await pyodide.loadPackage(['numpy', 'pandas', 'scipy']);
      return pyodide;
    })();
    return this.ready;
  }

  async run(code: string, tests: string): Promise<string> {
    const pyodide = await this.init();
    const buffer: string[] = [];
    pyodide.setStdout({ batched: (s: string) => buffer.push(s) });
    pyodide.setStderr({ batched: (s: string) => buffer.push('ERR: ' + s) });
    try {
      await pyodide.runPythonAsync(code);
      if (tests) {
        const res = pyodide.runPython(tests);
        buffer.push('\n[Tests] ' + (res ? 'PASSED' : 'FAILED'));
      }
      return buffer.join('\n') || '✓ No output';
    } catch (e: any) {
      return buffer.join('\n') + '\n' + e.message;
    }
  }
}

    10. 100+ Unique Question Generator — Parameterized Architecture

    Instead of hardcoding 100 static text questions per module, we use a Parameterized Procedural Question Generator. A single generator definition produces unlimited non-repeating variants by randomizing datasets, distributions, parameters, and edge cases. Each variant comes with starter code, auto-tests, and hints.

    components/editor/QuestionGenerator.ts
    // A single generator definition produces unlimited variants by randomizing parameters.
export type QVariant = { prompt: string; starter: string; tests: string; hint: string };

export interface QuestionGenerator {
  id: string;
  title: string;
  difficulty: 'easy' | 'medium' | 'hard';
  generate(seed?: number): QVariant;
}

// Helper: seeded RNG so variants are reproducible but non-repeating across sessions
export function rng(seed = Date.now()) {
  let s = seed % 2147483647;
  if (s <= 0) s += 2147483646;
  return () => (s = (s * 16807) % 2147483647) / 2147483647;
}

export function generateN(gens: QuestionGenerator[], n: number): QVariant[] {
  const pool: QVariant[] = [];
  for (let i = 0; i < n; i++) {
    const g = gens[Math.floor(Math.random() * gens.length)];
    pool.push(g.generate(i * 7919 + 1337));
  }
  return pool;
}

    lib/llm.ts — BYOK OpenAI-compatible client (also works with Ollama)
    import { storage } from './storage';

export async function llmChat(messages: { role: string; content: string }[]): Promise<string> {
  const cfg: any = await storage.loadKey();
  const apiKey = cfg?.apiKey || process.env.NEXT_PUBLIC_OPENAI_KEY;
  const endpoint = cfg?.endpoint || 'https://api.openai.com/v1/chat/completions';

  if (!apiKey) return "Please add an API key in Settings, or install Ollama locally.";

  const res = await fetch(endpoint, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${apiKey}` },
    body: JSON.stringify({ model: cfg?.model || 'gpt-4o-mini', messages, temperature: 0.7, max_tokens: 800 })
  });
  const data = await res.json();
  return data.choices?.[0]?.message?.content || 'No response.';
}
